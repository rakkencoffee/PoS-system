========== 5. OLSERA TOKEN AUTH (3-LAYER CACHE) ==========

flowchart TD
    START(["getAccessToken called"]) --> CHECK_MEMORY{"Layer 1:\nIn-Memory Cache\n~0ms"}

    CHECK_MEMORY -->|"Token valid\n(not expired)"| RETURN_TOKEN["Return Token"]
    CHECK_MEMORY -->|"null / expired"| CHECK_REDIS{"Layer 2:\nUpstash Redis\n~5ms"}

    CHECK_REDIS -->|"Token found\n(not expired)"| WARM_MEMORY["Warm In-Memory Cache"]
    WARM_MEMORY --> RETURN_TOKEN
    CHECK_REDIS -->|"null / expired"| CHECK_MUTEX{"Mutex Check:\ntokenRefreshPromise?"}

    CHECK_MUTEX -->|"Already fetching"| WAIT_MUTEX["Wait for existing\nfetch to complete"]
    WAIT_MUTEX --> RETURN_TOKEN

    CHECK_MUTEX -->|"No pending fetch"| FETCH_TOKEN["Layer 3:\nFetch from Olsera API\n~500-2000ms"]

    FETCH_TOKEN --> CHECK_CREDS{"Credentials\nPresent?"}
    CHECK_CREDS -->|"Missing"| THROW_ERROR["Throw:\nCredentials not configured"]

    CHECK_CREDS -->|"Present"| RETRY_LOOP["Retry Loop\n(max 3 attempts,\nexponential backoff)"]

    RETRY_LOOP --> ATTEMPT{"POST /api/open-api/\nv1/id/token\n(10s timeout)"}

    ATTEMPT -->|"200 OK"| EXTRACT_TOKEN["Extract access_token"]
    EXTRACT_TOKEN --> SAVE_MEMORY["Save to In-Memory"]
    SAVE_MEMORY --> SAVE_REDIS["Save to Redis\n(TTL: 50 min)"]
    SAVE_REDIS --> RETURN_TOKEN

    ATTEMPT -->|"401"| RETRY_401["Retry with backoff"]
    ATTEMPT -->|"500/Network"| RETRY_NET["Retry with backoff"]
    RETRY_401 --> RETRY_LOOP
    RETRY_NET --> RETRY_LOOP

    ATTEMPT -->|"All 3 failed"| THROW_FINAL["Throw: Auth failed\nafter all retries"]

    subgraph ON_401["On 401 During API Call"]
        INVALIDATE_MEM["Clear In-Memory"]
        INVALIDATE_REDIS["redis.del token key"]
        RETRY_CALL["Retry API Call\n(with fresh token)"]
    end

    INVALIDATE_MEM --> INVALIDATE_REDIS
    INVALIDATE_REDIS --> RETRY_CALL

========== 7. ORDER STATUS STATE MACHINE ==========

stateDiagram-v2
    [*] --> PENDING: Customer places order

    PENDING --> PREPARING: Barista starts making
    PENDING --> CANCELLED: Customer cancels

    PREPARING --> READY: Order completed by barista
    
    READY --> COMPLETED: Customer picks up

    COMPLETED --> [*]
    CANCELLED --> [*]

    state PENDING {
        [*] --> WaitingPayment
        WaitingPayment --> PaymentReceived: Midtrans webhook
    }

    note right of PENDING
        Olsera Status: P
        Kiosk: Order placed
        KDS: Yellow card
    end note

    note right of PREPARING
        Olsera Status: A
        KDS: Orange card
        Barista making drinks
    end note

    note right of READY
        Olsera Status: S
        KDS: Green card
        Order Ready notification
    end note

    note right of COMPLETED
        Olsera Status: Z
        Archived from KDS
        Synced to Olsera
    end note

    note right of CANCELLED
        Olsera Status: X
        Refund if paid
    end note
